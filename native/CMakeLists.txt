cmake_minimum_required(VERSION 3.24)
project(NexonicTorrent VERSION 1.0.0 LANGUAGES CXX)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
option(NDM_TORRENT_STATIC "Build a static Apple/mobile bridge" OFF)
option(NDM_TORRENT_TESTS "Build legal-data native integration tests" ON)
find_package(LibtorrentRasterbar CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(OpenSSL REQUIRED)
set(NDM_LIBRARY_KIND SHARED)
if(NDM_TORRENT_STATIC)
  set(NDM_LIBRARY_KIND STATIC)
endif()
add_library(nexonic_torrent ${NDM_LIBRARY_KIND} src/engine.cpp)
target_compile_features(nexonic_torrent PUBLIC cxx_std_17)
if(ANDROID)
  target_sources(nexonic_torrent PRIVATE src/android_jni.cpp)
  target_link_options(nexonic_torrent PRIVATE "-Wl,-z,max-page-size=16384")
endif()
target_include_directories(nexonic_torrent PUBLIC include)
target_link_libraries(nexonic_torrent PRIVATE LibtorrentRasterbar::torrent-rasterbar nlohmann_json::nlohmann_json OpenSSL::Crypto)
target_compile_definitions(nexonic_torrent PRIVATE NDM_TORRENT_EXPORTS NOMINMAX)
if(NDM_TORRENT_STATIC)
  target_compile_definitions(nexonic_torrent PUBLIC NDM_TORRENT_STATIC)
endif()
set_target_properties(nexonic_torrent PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(MSVC)
  target_compile_options(nexonic_torrent PRIVATE /W4 /EHsc /utf-8)
else()
  target_compile_options(nexonic_torrent PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS nexonic_torrent ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(FILES include/nexonic_torrent.h DESTINATION include)
if(NDM_TORRENT_TESTS)
  enable_testing()
  add_executable(torrent_native_test tests/engine_test.cpp)
  target_compile_features(torrent_native_test PRIVATE cxx_std_17)
  target_link_libraries(torrent_native_test PRIVATE nexonic_torrent nlohmann_json::nlohmann_json OpenSSL::Crypto)
  add_test(NAME native_legal_torrent_transfer COMMAND torrent_native_test)
  set_tests_properties(native_legal_torrent_transfer PROPERTIES TIMEOUT 120)
endif()
